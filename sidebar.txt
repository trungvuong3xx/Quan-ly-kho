<script>
(function () {
  'use strict';

  const Api = {
    call(fnName, ...args) {
      return new Promise((resolve, reject) => {
        google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(err => reject(err instanceof Error ? err : new Error(String(err && err.message || err))))
          [fnName](...args);
      });
    },

    fireAndForget(fnName, ...args) {
      google.script.run
        .withFailureHandler(err => console.warn('[fireAndForget] ' + fnName + ' lỗi:', err))
        [fnName](...args);
    }
  };

  const Utils = {
    boDau(str) {
      return str.normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .replace(/đ/g, 'd')
        .replace(/Đ/g, 'D');
    },
    todayStr() {
      const d = new Date();
      const m = String(d.getMonth() + 1).padStart(2, '0');
      const day = String(d.getDate()).padStart(2, '0');
      return d.getFullYear() + '-' + m + '-' + day;
    },
    shortDate(dateStr) {
      const p = dateStr.split('-');
      return p.length === 3 ? p[2] + '/' + p[1] : dateStr;
    },

    toX10(n) { return Math.round((parseFloat(n) || 0) * 10); },
    formatX10(x10) { return (x10 / 10).toString(); },
    escapeHtml(str) {
      return String(str).replace(/[&<>"']/g, c => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
      }[c]));
    }
  };

  const Toast = {
    el: null,
    init() { this.el = document.getElementById('toastContainer'); },
    show(type, message, duration = 4000) {
      const item = document.createElement('div');
      item.className = 'toast toast-' + type;
      item.textContent = message;
      this.el.appendChild(item);
      requestAnimationFrame(() => item.classList.add('show'));
      setTimeout(() => {
        item.classList.remove('show');
        setTimeout(() => item.remove(), 220);
      }, duration);
    },
    ok(msg) { this.show('ok', msg); },
    err(msg) { this.show('err', msg, 5000); }
  };

  const State = {
    mspData: [],
    filteredList: [],
    activeIndex: -1,
    selectedVung: 'NORMAL',
    historyList: [],
    historySeq: 0,
    sessionStats: { count: 0, baoX10: 0, kgX10: 0 },
    pendingCount: 0
  };

  const Dom = {};
  function cacheDom() {
    [
      'ngay', 'loai', 'ghiChu', 'ten', 'tenDropdown', 'tenLoadingIcon', 'msp', 'vungIndicator',
      'bao', 'kg', 'btnGhi', 'btnGhiLabel', 'btnGhiBadge', 'btnLamMoi', 'titleSpinner',
      'statCount', 'statBao', 'statKg', 'historyBody',
      'mainView', 'pageLichSuDaLuu', 'btnMoLichSuDaLuu', 'btnQuayLai', 'lsdlNgay', 'btnTimLichSu', 'lsdlBody'
    ].forEach(id => { Dom[id] = document.getElementById(id); });
  }

  const Dropdown = {
    isLoaiXuat() { return Dom.loai.value.indexOf('XUAT_') === 0; },

    filter(query) {
      const q = Utils.boDau(query.trim()).toUpperCase();
      if (!q) return [];

      const matched = State.mspData.filter(item => Utils.boDau(item.ten).toUpperCase().includes(q));

      let result;
      if (!this.isLoaiXuat()) {
        const normalMspSet = new Set(matched.filter(i => i.vung === 'NORMAL').map(i => i.msp));
        result = matched.filter(item => item.vung !== 'FOR' || !normalMspSet.has(item.msp));
      } else {
        result = matched;
      }

      result = result.slice().sort((a, b) => (b.count || 0) - (a.count || 0));
      return result.slice(0, 50);
    },

    render() {
      const el = Dom.tenDropdown;
      if (State.filteredList.length === 0) {
        el.classList.remove('open');
        el.innerHTML = '';
        return;
      }
      el.innerHTML = State.filteredList.map((item, idx) =>
        '<div class="ten-dropdown-item' + (idx === State.activeIndex ? ' active' : '') + '" data-idx="' + idx + '">' +
        Utils.escapeHtml(item.ten) + '</div>'
      ).join('');
      el.classList.add('open');

      Array.from(el.children).forEach(child => {
        child.addEventListener('mousedown', e => {
          e.preventDefault(); // tránh ô input bị blur trước khi click được ghi nhận
          this.select(State.filteredList[parseInt(child.getAttribute('data-idx'), 10)]);
        });
      });
    },

    close() {
      State.filteredList = [];
      State.activeIndex = -1;
      Dom.tenDropdown.classList.remove('open');
      Dom.tenDropdown.innerHTML = '';
    },

    select(item) {
      Dom.ten.value = item.ten;
      Dom.msp.value = item.msp;
      State.selectedVung = item.vung || 'NORMAL';
      this.updateVungIndicator();
      this.close();
      Dom.bao.focus();
      Dom.bao.select();
    },

    updateVungIndicator() {
      const el = Dom.vungIndicator;
      if (!this.isLoaiXuat()) {
        el.className = 'vung-indicator';
        el.textContent = '';
        return;
      }
      if (State.selectedVung === 'FOR') {
        el.className = 'vung-indicator for';
        el.textContent = 'Đang chọn: Chỉ X1';
      } else {
        el.className = 'vung-indicator thuong';
        el.textContent = 'Đang chọn: Chỉ X5';
      }
    },

    scrollActiveIntoView() {
      const active = Dom.tenDropdown.querySelector('.ten-dropdown-item.active');
      if (active) active.scrollIntoView({ block: 'nearest' });
    },

    onInput() {
      Dom.msp.value = '';
      State.filteredList = this.filter(Dom.ten.value);
      State.activeIndex = -1;
      this.render();
    },

    onKeydown(e) {
      const list = State.filteredList;
      if (e.key === 'ArrowDown') {
        if (!list.length) return;
        e.preventDefault();
        State.activeIndex = Math.min(State.activeIndex + 1, list.length - 1);
        this.render();
        this.scrollActiveIntoView();
      } else if (e.key === 'ArrowUp') {
        if (!list.length) return;
        e.preventDefault();
        State.activeIndex = Math.max(State.activeIndex - 1, 0);
        this.render();
        this.scrollActiveIntoView();
      } else if (e.key === 'Enter') {
        e.preventDefault();
        if (list.length) this.select(list[State.activeIndex >= 0 ? State.activeIndex : 0]);
        else Toast.err('Làm gì có QC này???');
      } else if (e.key === 'Tab' && !e.shiftKey) {
        if (list.length) {
          e.preventDefault();
          this.select(list[State.activeIndex >= 0 ? State.activeIndex : 0]);
        } else {
          e.preventDefault();
          Toast.err('Làm gì có QC này???');
        }
      } else if (e.key === 'Escape') {
        if (list.length) {
          e.stopPropagation(); 
          this.close();
        }
      }
    }
  };

  const SessionSummary = {
    render() {
      Dom.statCount.textContent = State.sessionStats.count;
      Dom.statBao.textContent = Utils.formatX10(State.sessionStats.baoX10);
      Dom.statKg.textContent = Utils.formatX10(State.sessionStats.kgX10);
    }
  };

  const History = {
    SLOW_THRESHOLD_MS: 15000,

    render() {
      Dom.historyBody.innerHTML = State.historyList.map(item => {
        let cls = '';
        let icon = '';
        let title = item.message ? Utils.escapeHtml(item.message) : '';

        if (item.status === 'pending' && item.slow) {
          cls = 'row-slow';
          icon = ' ⏳⚠';
          title = 'Đang ghi lâu bất thường, vui lòng chờ...';
        } else if (item.status === 'pending') {
          cls = 'row-pending';
          icon = ' ⏳';
        } else if (item.status === 'err') {
          cls = 'row-err';
          icon = ' ⚠';
        }

        return '<tr class="' + cls + '" title="' + title + '">' +
          '<td>' + item.ngay + '</td>' +
          '<td>' + item.loai + '</td>' +
          '<td>' + Utils.escapeHtml(item.ten) + icon + '</td>' +
          '<td>' + item.bao + '</td>' +
          '<td>' + item.kg + '</td>' +
          '</tr>';
      }).join('');
    },

    addPending(dateStr, loaiText, tenSp, bao, kg) {
      const id = ++State.historySeq;
      const item = {
        id, ngay: Utils.shortDate(dateStr), loai: loaiText, ten: tenSp,
        bao: bao || '0', kg: kg || '0', status: 'pending', message: '', slow: false, slowTimer: null
      };
      State.historyList.unshift(item);

      // Hẹn giờ: nếu sau 15s vẫn chưa có phản hồi (còn "pending"), bật cờ cảnh báo
      item.slowTimer = setTimeout(() => {
        if (item.status === 'pending') {
          item.slow = true;
          this.render();
        }
      }, this.SLOW_THRESHOLD_MS);

      this.render();
      return id;
    },

    updateStatus(id, status, message) {
      const item = State.historyList.find(h => h.id === id);
      if (!item) return; // đã bị đẩy khỏi danh sách hiển thị, bỏ qua
      if (item.slowTimer) { clearTimeout(item.slowTimer); item.slowTimer = null; }

      const wasOk = item.status === 'ok';
      item.status = status;
      item.message = message || '';
      item.slow = false; // đã có kết quả, không còn cảnh báo "đang ghi lâu" nữa

      if (status === 'ok' && !wasOk) {
        State.sessionStats.count += 1;
        State.sessionStats.baoX10 += Utils.toX10(item.bao);
        State.sessionStats.kgX10 += Utils.toX10(item.kg);
        SessionSummary.render();
      }
      this.render();
    }
  };

  const LichSuDaLuu = {
    open() {
      Dom.mainView.style.display = 'none';
      Dom.pageLichSuDaLuu.style.display = 'block';
      Dom.lsdlNgay.value = Dom.ngay.value || Utils.todayStr();
      Dom.lsdlBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--text-muted); font-style: italic;">Chọn ngày và bấm tìm</td></tr>';
    },

    close() {
      Dom.pageLichSuDaLuu.style.display = 'none';
      Dom.mainView.style.display = 'block';
    },

    timKiem() {
      const dateStr = Dom.lsdlNgay.value;
      if (!dateStr) { Toast.err('Chưa chọn ngày'); return; }

      Dom.btnTimLichSu.disabled = true;
      Dom.lsdlBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--text-muted); font-style: italic;">Đang tìm...</td></tr>';

      Api.call('layLichSu', dateStr)
        .then(list => {
          Dom.btnTimLichSu.disabled = false;
          if (!list || list.length === 0) {
            Dom.lsdlBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--text-muted); font-style: italic;">Không có lượt nào trong ngày này</td></tr>';
            return;
          }
          Dom.lsdlBody.innerHTML = list.map(item =>
            '<tr>' +
              '<td>' + Utils.shortDate(dateStr) + '</td>' +
              '<td>' + Utils.escapeHtml(item.loai) + '</td>' +
              '<td>' + Utils.escapeHtml(item.ten) + '</td>' +
              '<td>' + item.bao + '</td>' +
              '<td>' + item.kg + '</td>' +
            '</tr>'
          ).join('');
        })
        .catch(err => {
          Dom.btnTimLichSu.disabled = false;
          Dom.lsdlBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--err-text);">Lỗi tải dữ liệu</td></tr>';
          Toast.err('Lỗi tải lịch sử: ' + err.message);
        });
    }
  };

  const Queue = {
    cellQueues: new Map(), // cellKey -> Promise (chuỗi tuần tự riêng cho từng ô)

    updateBadge() {
      if (State.pendingCount > 0) {
        Dom.btnGhiBadge.style.display = 'inline-block';
        Dom.btnGhiBadge.textContent = State.pendingCount;
      } else {
        Dom.btnGhiBadge.style.display = 'none';
      }
    },

    keyFor(payload) {
      return [payload.msp, payload.dateStr, payload.loai, payload.vung].join('|');
    },

    push(payload, job) {
      const key = this.keyFor(payload);
      const prevInCell = this.cellQueues.get(key) || Promise.resolve();

      State.pendingCount++;
      this.updateBadge();

      const nextInCell = prevInCell
        .then(job)
        .catch(err => console.error('[Queue] lỗi không mong muốn tại ô ' + key + ':', err))
        .finally(() => {
          State.pendingCount = Math.max(0, State.pendingCount - 1);
          this.updateBadge();
          // Dọn dẹp: nếu không còn lượt nào khác đang chờ nối tiếp ô này, xóa khỏi map
          if (this.cellQueues.get(key) === nextInCell) this.cellQueues.delete(key);
        });

      this.cellQueues.set(key, nextInCell);
    }
  };

  const FormModule = {
    validate(payload) {
      if (!payload.dateStr) return 'Chưa chọn ngày';
      if (!payload.msp) return 'Làm gì có QC này???';
      if (!payload.bao && !payload.kg) return 'Chưa nhập Bao hoặc Kg';
      const bao = Number(payload.bao) || 0;
      const kg = Number(payload.kg) || 0;
      if (bao < 0 || kg < 0) return 'Bao/Kg không thể âm';
      return null;
    },

    getPayload() {
      const loaiSelect = Dom.loai;
      return {
        dateStr: Dom.ngay.value,
        loai: loaiSelect.value,
        loaiText: loaiSelect.options[loaiSelect.selectedIndex].text,
        ten: Dom.ten.value,
        msp: Dom.msp.value,
        bao: Dom.bao.value,
        kg: Dom.kg.value,
        vung: State.selectedVung,
        ghiChu: Dom.ghiChu.value
      };
    },

    reset() {
      Dom.ten.value = '';
      Dom.msp.value = '';
      Dom.bao.value = '';
      Dom.kg.value = '';
      State.selectedVung = 'NORMAL';
      Dom.vungIndicator.className = 'vung-indicator';
      Dom.vungIndicator.textContent = '';
      Dropdown.close();
    },

    resetGhiChu() {
      Dom.ghiChu.value = '';
    },

    ghi() {
      const p = this.getPayload();
      const error = this.validate(p);
      if (error) { Toast.err(error); return; }

      // Hiện ngay dòng "đang ghi" trong lịch sử, xóa trắng form để gõ lượt tiếp NGAY LẬP TỨC
      const historyId = History.addPending(p.dateStr, p.loaiText, p.ten, p.bao, p.kg);
      this.reset();
      Dom.ten.focus();

      const payload = { dateStr: p.dateStr, loai: p.loai, msp: p.msp, bao: p.bao, kg: p.kg, vung: p.vung, ghiChu: p.ghiChu };

      Queue.push(payload, () =>
        Api.call('submitEntry', payload)
          .then(res => {
            if (res.success) {
              History.updateStatus(historyId, 'ok', res.message);
              Api.fireAndForget('capNhatSoLanDung', payload.msp, res.vung || payload.vung, payload.dateStr);
              Api.fireAndForget('ghiLichSu', {
                dateStr: payload.dateStr,
                loaiText: p.loaiText,
                ten: p.ten,
                msp: payload.msp,
                bao: payload.bao,
                kg: payload.kg,
                vung: res.vung || payload.vung
              });
              const localItem = State.mspData.find(m => m.msp === payload.msp && (m.vung || 'NORMAL') === (res.vung || payload.vung));
              if (localItem) localItem.count = (localItem.count || 0) + 1;
            } else {
              History.updateStatus(historyId, 'err', res.message);
              Toast.err(p.ten + ': ' + res.message);
            }
          })
          .catch(err => {
            History.updateStatus(historyId, 'err', 'Lỗi hệ thống: ' + err.message);
            Toast.err(p.ten + ': Lỗi hệ thống - ' + err.message);
          })
      );
    }
  };

  const DataLoader = {
    requestToken: 0,
    monthCache: {}, // cache trong phiên (bộ nhớ) — key: tên sheet tháng -> { exists, sheetName, mspList }

    setLoading(loading) {
      Dom.ten.disabled = loading;
      Dom.ten.placeholder = loading ? 'Đang tải danh sách...' : 'Gõ để tìm...';
      Dom.tenLoadingIcon.classList.toggle('show', loading);
      Dom.titleSpinner.style.display = loading ? 'inline-block' : 'none';
    },

    monthKeyFor(dateStr) {
      const d = new Date(dateStr);
      if (isNaN(d.getTime())) return null;
      return String(d.getMonth() + 1);
    },

    applyResult(res) {
      if (res.error) {
        Toast.err(res.error);
      } else if (!res.exists) {
        Toast.err('Chưa có sheet tháng "' + res.sheetName + '". Vui lòng tạo sheet trước khi ghi.');
      }
      State.mspData = res.mspList || [];
    },

    // Gộp kiemTraThangSheet + getMspList thành 1 round-trip duy nhất qua khoiTaoForm.
    // Cache trong bộ nhớ (monthCache) chỉ sống trong lúc sidebar đang mở — đổi ngày
    // cùng tháng trong phiên hiện tại sẽ không phải gọi lại server.
    // Lưu ý: localStorage KHÔNG dùng được ở đây vì mỗi lần mở sidebar, Apps Script
    // gán cho iframe 1 domain sandbox khác nhau, nên không có tầng cache nào sống
    // được qua các lần mở lại — mỗi lần mở form đều cần ít nhất 1 round-trip.
    initForDate(dateStr) {
      const monthKey = this.monthKeyFor(dateStr);

      if (monthKey && this.monthCache[monthKey]) {
        FormModule.reset();
        this.applyResult(this.monthCache[monthKey]);
        Dom.ten.focus();
        return Promise.resolve();
      }

      const myToken = ++this.requestToken;
      this.setLoading(true);
      FormModule.reset();

      return Api.call('khoiTaoForm', dateStr, false)
        .then(res => {
          if (myToken !== this.requestToken) return;

          if (monthKey && !res.error) {
            this.monthCache[monthKey] = res;
          }

          this.applyResult(res);
          this.setLoading(false);
          Dom.ten.focus();
        })
        .catch(err => {
          if (myToken !== this.requestToken) return;
          this.setLoading(false);
          Toast.err('Lỗi tải dữ liệu: ' + err.message);
        });
    },

    onDateChange() {
      const dateStr = Dom.ngay.value;
      FormModule.resetGhiChu();
      this.initForDate(dateStr);
    },

    // Ép tính lại từ đầu, bỏ qua cache server (cache bộ nhớ client cũng bị xóa theo).
    refreshCurrentMonth() {
      const dateStr = Dom.ngay.value;
      const monthKey = this.monthKeyFor(dateStr);
      if (!monthKey) { Toast.err('Chưa chọn ngày hợp lệ'); return; }

      delete this.monthCache[monthKey];

      const myToken = ++this.requestToken;
      Dom.btnLamMoi.disabled = true;
      Dom.btnLamMoi.classList.add('loading');
      this.setLoading(true);
      FormModule.reset();

      Api.call('khoiTaoForm', dateStr, true)
        .then(res => {
          if (myToken !== this.requestToken) return;

          if (!res.error) {
            this.monthCache[monthKey] = res;
          }

          this.applyResult(res);
          this.setLoading(false);
          Dom.btnLamMoi.disabled = false;
          Dom.btnLamMoi.classList.remove('loading');
          Dom.ten.focus();
          if (!res.error) Toast.ok('Đã làm mới danh sách');
        })
        .catch(err => {
          if (myToken !== this.requestToken) return;
          this.setLoading(false);
          Dom.btnLamMoi.disabled = false;
          Dom.btnLamMoi.classList.remove('loading');
          Toast.err('Lỗi làm mới: ' + err.message);
        });
    }
  };

  const App = {
    init() {
      cacheDom();
      Toast.init();

      Dom.ngay.value = Utils.todayStr();

      Dom.ten.addEventListener('input', () => Dropdown.onInput());
      Dom.ten.addEventListener('keydown', e => Dropdown.onKeydown(e));

      document.addEventListener('click', e => {
        if (!e.target.closest('.ten-wrap')) Dropdown.close();
      });

      const submitOnEnter = e => { if (e.key === 'Enter') { e.preventDefault(); FormModule.ghi(); } };
      const blockMinus = e => { if (e.key === '-' || e.key === 'e') e.preventDefault(); };

      Dom.bao.addEventListener('keydown', submitOnEnter);
      Dom.bao.addEventListener('keydown', blockMinus);
      Dom.kg.addEventListener('keydown', submitOnEnter);
      Dom.kg.addEventListener('keydown', blockMinus);
      Dom.btnGhi.addEventListener('click', () => FormModule.ghi());
      Dom.btnLamMoi.addEventListener('click', () => DataLoader.refreshCurrentMonth());
      Dom.btnMoLichSuDaLuu.addEventListener('click', () => LichSuDaLuu.open());
      Dom.btnQuayLai.addEventListener('click', () => LichSuDaLuu.close());
      Dom.btnTimLichSu.addEventListener('click', () => LichSuDaLuu.timKiem());

      window.addEventListener('keydown', e => {
        if (e.key === 'Escape') {
          FormModule.reset();
          Dom.ten.focus();
        }
      });

      Dom.ngay.addEventListener('change', () => DataLoader.onDateChange());
      Dom.loai.addEventListener('change', () => {
        FormModule.reset();
        FormModule.resetGhiChu();
      });

      DataLoader.onDateChange();
    }
  };

  App.init();
})();
</script>
