const DATA_START_ROW = 6;
const COL_ID = 1;
const COL_MSP = 2;
const COL_QC = 3;
const COL_BAO = 4;
const COL_KG = 5;
const COL_LOAI = 6;
const COL_NGAY = 7;
const COL_THOIGIAN = 8;

function getInfo(msp) {
  try {
    const cleanMsp = String(msp || "").trim();
    if (!cleanMsp) return { success: false, message: "Thiếu MSP" };

    let map = null;
    const cache = CacheService.getScriptCache();
    try {
      const cached = cache.get("DANHMUC_MSP_MAP_V1");
      if (cached) map = JSON.parse(cached);
    } catch (eCache) {}

    if (!map) {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sh = ss.getSheetByName("DanhMuc");
      if (!sh) return { error: "Không tìm thấy sheet 'DanhMuc'" };

      const data = sh.getDataRange().getValues();
      map = {};
      for (let i = 3; i < data.length; i++) {
        const k = String(data[i][0] || "").trim();
        if (k) {
          map[k] = { ten: data[i][1], mau: data[i][2] };
        }
      }
      try { cache.put("DANHMUC_MSP_MAP_V1", JSON.stringify(map), 600); } catch (ePut) {}
    }

    if (map[cleanMsp]) {
      return { success: true, ten: map[cleanMsp].ten, mau: map[cleanMsp].mau };
    }

    return { success: false, message: "Không tìm thấy MSP: " + msp };
  } catch (e) {
    return { error: e.message };
  }
}

function taoNhieuID(soLuong) {
  try {
    const existingIds = new Set();
    try {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const shNhap = ss.getSheetByName("NHẬP");
      if (shNhap && shNhap.getLastRow() >= DATA_START_ROW) {
        const colIds = shNhap.getRange(DATA_START_ROW, COL_ID, shNhap.getLastRow() - DATA_START_ROW + 1, 1).getValues();
        for (let i = 0; i < colIds.length; i++) {
          const v = String(colIds[i][0] || "").trim();
          if (v) existingIds.add(v);
        }
      }
    } catch (eRead) {}

    const ids = [];
    for (let i = 0; i < soLuong; i++) {
      let id;
      let attempts = 0;
      do {
        const so = Math.floor(Math.random() * 90000) + 10000;
        id = "5C" + so;
        attempts++;
      } while ((ids.includes(id) || existingIds.has(id)) && attempts < 100000);
      ids.push(id);
      existingIds.add(id);
    }
    return ids;
  } catch(e) {
    return { error: e.message };
  }
}

function isNhap(loai) {
  return String(loai || "").trim().startsWith("Nhập");
}

function isXuat(loai) {
  const text = String(loai || "").trim();
  return text.startsWith("Xuất") || text === "Bình Thuận";
}

function layTenSheet(loai) {
  const text = String(loai || "").trim();
  if (isNhap(text)) return "NHẬP";
  if (text === "Bình Thuận") return "BÌNH THUẬN";
  if (text.toUpperCase() === "XUẤT FOR") return "FOR";
  if (text.startsWith("Xuất")) return "XUẤT";
  return null;
}

function parseKg(value) {
  if (value === "" || value === null || value === undefined) return 0;
  const num = parseFloat(String(value).replace(",", "."));
  return Number.isFinite(num) ? num : 0;
}

function lamTronKg(value) {
  return Math.round((parseFloat(value) || 0) * 1000) / 1000;
}

function formatNgay(dateVal) {
  if (!dateVal) return "";
  return Utilities.formatDate(new Date(dateVal), Session.getScriptTimeZone(), "dd/MM/yyyy");
}

function formatGio(dateVal) {
  if (!dateVal) return "";
  return Utilities.formatDate(new Date(dateVal), Session.getScriptTimeZone(), "HH:mm:ss");
}

function getDataRows(sheet) {
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < DATA_START_ROW) return [];

  return sheet
    .getRange(DATA_START_ROW, 1, lastRow - DATA_START_ROW + 1, COL_THOIGIAN)
    .getValues()
    .map((row, index) => ({ rowNumber: DATA_START_ROW + index, values: row }))
    .filter(item => String(item.values[COL_ID - 1]).trim() !== "");
}

function timRowsTheoIdNhanh_(sheet, targetId) {
  if (!sheet) return [];
  const cleanId = String(targetId || "").trim();
  if (!cleanId) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < DATA_START_ROW) return [];

  try {
    const range = sheet.getRange(DATA_START_ROW, COL_ID, lastRow - DATA_START_ROW + 1, 1);
    const matches = range.createTextFinder(cleanId).matchEntireCell(true).findAll();
    if (!matches || matches.length === 0) return [];
    return matches.map(cell => {
      const rNum = cell.getRow();
      const rowVals = sheet.getRange(rNum, 1, 1, COL_THOIGIAN).getValues()[0];
      return { rowNumber: rNum, values: rowVals };
    });
  } catch (e) {
    return getDataRows(sheet).filter(row => rowId(row) === cleanId);
  }
}

function rowId(row) {
  return String(row.values[COL_ID - 1]).trim();
}

function rowKg(row) {
  const kg = row.values[COL_KG - 1];
  return parseKg(kg);
}

function thongTinTon(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const shNhap = ss.getSheetByName("NHẬP");
  const shXuat = ss.getSheetByName("XUẤT");
  const shBinhThuan = ss.getSheetByName("BÌNH THUẬN");
  const shFor = ss.getSheetByName("FOR");

  const nhapRows = timRowsTheoIdNhanh_(shNhap, id);
  const xuatRows = []
    .concat(timRowsTheoIdNhanh_(shXuat, id))
    .concat(timRowsTheoIdNhanh_(shBinhThuan, id))
    .concat(timRowsTheoIdNhanh_(shFor, id));

  const tongNhap = lamTronKg(nhapRows.reduce((sum, row) => sum + rowKg(row), 0));
  const tongXuat = lamTronKg(xuatRows.reduce((sum, row) => sum + rowKg(row), 0));
  const ton = lamTronKg(tongNhap - tongXuat);
  
  const nhapRow = nhapRows.length ? nhapRows[0] : null;

  return {
    tongNhap,
    tongXuat,
    ton,
    kgNhap: nhapRow ? rowKg(nhapRow) : 0
  };
}

function kiemTraQR(id, msp, loai) {
  try {
    const info = getInfo(msp);
    if (info.error) return { error: info.error };

    const tonInfo = thongTinTon(id);
    const base = {
      success: true,
      ten: info.success ? info.ten : "",
      mau: info.success ? info.mau : "",
      tongNhap: tonInfo.tongNhap,
      tongXuat: tonInfo.tongXuat,
      ton: tonInfo.ton,
      kgNhap: tonInfo.kgNhap
    };

    if (isNhap(loai)) {
      base.cheDo = "nhapMoi"; // Xóa cơ chế ghi đè, luôn trả về nhapMoi
      return base;
    }

    if (isXuat(loai)) {
      if (tonInfo.tongNhap <= 0) return { error: "Mã chưa tồn" };
      if (tonInfo.ton <= 0) return { error: "Mã đã hết tồn" };
      base.cheDo = "xuat";
      return base;
    }

    return { error: "Không xác định loại" };
  } catch (e) {
    return { error: e.message };
  }
}

function ghiDong(sheet, id, msp, qc, bao, kg, loai, ngay, thoiGian) {
  const dNgay = formatNgay(ngay);
  const dGio = formatGio(thoiGian ? thoiGian : new Date());
  
  sheet.appendRow([
    id,
    msp,
    qc,
    bao,
    kg,
    loai,
    dNgay,
    dGio
  ]);
}

function luuGiaoDich(id, msp, ten, mau, ngay, loai, kg, thoiGian) {
  const lock = LockService.getDocumentLock();
  try {
    lock.waitLock(30000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const tenSheet = layTenSheet(loai);
    if (!tenSheet) return { error: "Không xác định loại" };

    const sh = ss.getSheetByName(tenSheet);
    if (!sh) return { error: "Không tìm thấy sheet '" + tenSheet + "'" };

    const kgGoc = lamTronKg(parseKg(kg));
    if (kgGoc <= 0) return { error: "Kg không hợp lệ" };

    const tonInfo = thongTinTon(id);
    const qc = ten; // Ráp biến ten truyền từ app (r.qc) làm cột QC

    if (isNhap(loai)) {
      ghiDong(sh, id, msp, qc, 1, kgGoc, loai, ngay, thoiGian);
      return {
        success: true,
        cheDo: "nhapMoi",
        kgLuu: kgGoc,
        kgGoc: kgGoc,
        tongNhap: lamTronKg(tonInfo.tongNhap + kgGoc),
        tongXuat: tonInfo.tongXuat,
        ton: lamTronKg(tonInfo.tongNhap + kgGoc - tonInfo.tongXuat)
      };
    }

    if (isXuat(loai)) {
      if (tonInfo.tongNhap <= 0) return { error: "Mã chưa tồn" };
      if (tonInfo.ton <= 0) return { error: "Mã đã hết tồn" };
      if (kgGoc > tonInfo.ton) return { error: "Không đủ tồn: " + tonInfo.ton + " kg" };

      ghiDong(sh, id, msp, qc, 1, kgGoc, loai, ngay, thoiGian);
      return {
        success: true,
        cheDo: "xuat",
        kgLuu: kgGoc,
        kgGoc: kgGoc,
        tongNhap: tonInfo.tongNhap,
        tongXuat: lamTronKg(tonInfo.tongXuat + kgGoc),
        ton: lamTronKg(tonInfo.ton - kgGoc)
      };
    }

    return { error: "Không xác định loại" };
  } catch (e) {
    return { error: e.message };
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

// ─── Lưu kiểm kê batch ─────────────────────────────────────
function luuKiemKe(data) {
  const lock = LockService.getDocumentLock();
  try {
    lock.waitLock(30000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName("KIỂM KÊ");
    if (!sh) return { error: "Không tìm thấy sheet 'KIỂM KÊ'" };

    if (data && Array.isArray(data) && data.length > 0) {
      const rowsToInsert = data.map(row => {
        const dNgay = formatNgay(row.ngay);
        const dGio = formatGio(row.thoiGian ? row.thoiGian : new Date());
        return [row.id, row.msp, row.ten, 1, "", "Kiểm Kê", dNgay, dGio];
      });
      const startRow = Math.max(sh.getLastRow() + 1, DATA_START_ROW);
      sh.getRange(startRow, 1, rowsToInsert.length, 8).setValues(rowsToInsert);
    }
    return { success: true };
  } catch(e) {
    return { error: e.message };
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

function luuCX1(data) {
  const lock = LockService.getDocumentLock();
  try {
    lock.waitLock(30000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName("CX1");
    if (!sh) return { error: "Không tìm thấy sheet 'CX1'" };

    if (data && Array.isArray(data) && data.length > 0) {
      const lastRow = sh.getLastRow();
      const startRow = Math.max(lastRow + 1, 5); // Đổ dữ liệu từ hàng 5 trở đi

      const rowsToInsert = data.map(r => {
        const msp = String(r.msp || r.MSP || "").trim();
        const qc = String(r.qc || r.QC || "").trim();
        // Lấy số bao an toàn từ bất kỳ biến nào truyền sang
        const bao = r.bao || r.Bao || r.soLuong || r.SoLuong || r.soBao || r.sl || 1;
        const kg = Math.round((parseFloat(r.kg || r.KG) || 0) * 10) / 10;
        
        let dNgay = "";
        const rawDate = r.date || r.Date || r.ngay || r.Ngay;
        if (rawDate) {
          const s = String(rawDate).trim();
          const parts = s.split("-");
          if (parts.length === 3 && parts[0].length === 4) {
            dNgay = parts[2] + "/" + parts[1] + "/" + parts[0];
          } else {
            dNgay = s;
          }
        } else {
          dNgay = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy");
        }

        return [msp, qc, bao, kg, dNgay];
      });

      // Đổ đúng từ Cột 2 (Cột B: MSP) đến Cột 6 (Cột F: Date)
      sh.getRange(startRow, 2, rowsToInsert.length, 5).setValues(rowsToInsert);
    }
    
    return { success: true };
  } catch(e) {
    return { error: "Lỗi ghi Sheet: " + e.message };
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

function layDanhSachQC() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName("LIST");
    if (!sh) sh = ss.getSheetByName("DanhMuc");
    if (!sh) return { error: "Không tìm thấy sheet 'LIST'" };

    const lastRow = sh.getLastRow();
    if (lastRow < 2) return { success: true, list: [] };

    // Cột 1 (A): Tên/QC, Cột 2 (B): Mã MSP từ hàng 2 trở đi
    const data = sh.getRange(2, 1, lastRow - 1, 2).getValues();
    const list = data
      .filter(r => String(r[0]).trim() !== "" || String(r[1]).trim() !== "")
      .map(r => ({
        ten: String(r[0]).trim(),
        msp: String(r[1]).trim()
      }));

    return { success: true, list: list };
  } catch (e) {
    return { error: e.message };
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return reply({ error: "Dữ liệu gửi lên trống" });
    }
    const body = JSON.parse(e.postData.contents);
    const action = body.action;

    if (action === "layDanhSachQC") return reply(layDanhSachQC());
    if (action === "getInfo") return reply(getInfo(body.msp));
    if (action === "taoNhieuID") return reply(taoNhieuID(body.soLuong));
    if (action === "kiemTraQR") return reply(kiemTraQR(body.id, body.msp, body.loai));
    if (action === "luuKiemKe") return reply(luuKiemKe(body.data));
    if (action === "luuCX1") return reply(luuCX1(body.data));

    if (action === "luuGiaoDich") {
      return reply(luuGiaoDich(
        body.id,
        body.msp,
        body.ten,
        body.mau,
        body.ngay,
        body.loai,
        body.kg,
        body.thoiGian // Nhận thêm thời gian truyền từ app
      ));
    }

    return reply({ error: "Không rõ action" });
  } catch (err) {
    return reply({ error: "Lỗi xử lý request: " + err.message });
  }
}

function reply(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doOptions(e) {
  return ContentService.createTextOutput("")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doGet(e) {
  const key = e.parameter.key;
  let role = "user";
  if (key === "admin") role = "admin";
  const html = HtmlService.createTemplateFromFile("index");
  html.role = role;
  return html.evaluate();
}
