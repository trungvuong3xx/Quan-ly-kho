// ── Google Apps Script Dành Riêng Cho Quét BTP ────────────────
function doPost(e) {
  var result;
  try {
    var body = JSON.parse(e.postData.contents);
    if (body.action === "luuBTP") {
      result = luuBTP(body.data || body.payload);
    } else {
      result = { success: false, message: "Action không hợp lệ: " + body.action };
    }
  } catch (err) {
    result = { success: false, message: "Lỗi server: " + err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function luuBTP(data) {
  try {
    if (!data || !Array.isArray(data) || data.length === 0) {
      return { success: false, message: "Không có dữ liệu BTP" };
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    // Lấy sheet tên QRBTP hoặc sheet đang chọn hiện tại
    var sheet = ss.getSheetByName("QRBTP") || ss.getActiveSheet();

    // Dữ liệu đổ từ hàng 5 trở đi (không tự thêm dòng tiêu đề)
    var startRow = Math.max(sheet.getLastRow() + 1, 5);

    var rows = data.map(function(item) {
      var ngay = item.ngay || "";
      
      // Định dạng chỉ lấy Giờ:Phút:Giây (HH:mm:ss) chính xác lúc quét
      var gioQuet = "";
      if (item.thoiGian) {
        try {
          var d = new Date(item.thoiGian);
          if (!isNaN(d.getTime())) {
            gioQuet = Utilities.formatDate(d, "GMT+7", "HH:mm:ss");
          } else {
            gioQuet = String(item.thoiGian);
          }
        } catch (e) {
          gioQuet = String(item.thoiGian);
        }
      }

      return [
        item.rawQR || item.id || "",   // Cột A: Mã QR Gốc
        item.msp || "",                // Cột B: Mã BTP
        Number(item.kg || 0),          // Cột C: Số Lượng
        ngay,                          // Cột D: Ngày (YYYY-MM-DD)
        gioQuet                        // Cột E: Giờ quét (HH:mm:ss)
      ];
    });

    // Ghi mảng dữ liệu trực tiếp từ startRow
    sheet.getRange(startRow, 1, rows.length, 5).setValues(rows);

    // Kích hoạt tự động tổng hợp sang Sheet TONG_HOP_BTP
    try {
      tongHopBTP();
    } catch (eTongHop) {
      Logger.log("Lỗi tổng hợp BTP: " + eTongHop.message);
    }

    return { success: true, message: "Đã lưu " + rows.length + " dòng từ hàng " + startRow + " & đã tổng hợp BTP thành công" };
  } catch (err) {
    return { success: false, message: "Lỗi lưu BTP: " + err.message };
  }
}

// Tạo Menu trên Google Sheet
function onOpen() {
  try {
    SpreadsheetApp.getUi()
      .createMenu('📊 Quản Lý BTP')
      .addItem('⚡ Tổng Hợp BTP (Tất Cả)', 'chayTongHopBTPTatCa')
      .addItem('🔍 Tổng Hợp BTP (Chọn Khoảng Hàng)', 'chayTongHopBTPKhoang')
      .addToUi();
  } catch (e) {}
}

function chayTongHopBTP() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetTH = ss.getSheetByName("Tổng Hợp");
  var sheetQR = ss.getSheetByName("QRBTP") || ss.getActiveSheet();
  
  var lastRowQR = sheetQR ? sheetQR.getLastRow() : 5;
  var fromRow = 5;
  var toRow = lastRowQR;

  if (sheetTH) {
    var valB2 = sheetTH.getRange("B2").getValue();
    var valB3 = sheetTH.getRange("B3").getValue();
    if (valB2 && !isNaN(valB2)) fromRow = parseInt(valB2, 10);
    if (valB3 && !isNaN(valB3)) toRow = parseInt(valB3, 10);
  }

  if (fromRow < 5) fromRow = 5;
  if (toRow < fromRow) toRow = lastRowQR;
  if (toRow < 5) toRow = 5;

  tongHopBTPSheet(fromRow, toRow);
}

function chayTongHopBTPTatCa() {
  chayTongHopBTP();
}

function chayTongHopBTPKhoang() {
  chayTongHopBTP();
}

function tongHopBTP() {
  chayTongHopBTP();
}

function tongHopBTPSheet(fromRow, toRow) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetQC = ss.getSheetByName("QC");
  var sheetQR = ss.getSheetByName("QRBTP") || ss.getActiveSheet();
  var sheetTH = ss.getSheetByName("Tổng Hợp");

  if (!sheetTH) {
    sheetTH = ss.insertSheet("Tổng Hợp");
  }
  
  if (!sheetQR) return;
  if (!fromRow || fromRow < 5) fromRow = 5;
  if (!toRow || toRow < fromRow) toRow = sheetQR.getLastRow();
  if (toRow < 5) return;

  // 1. Đọc dữ liệu tra cứu từ Sheet QC (Bắt đầu từ HÀNG 2 trở xuống)
  var mapQC = {};
  if (sheetQC) {
    var lastRowQC = sheetQC.getLastRow();
    if (lastRowQC >= 2) {
      var qcVals = sheetQC.getRange(2, 1, lastRowQC - 1, 30).getValues();
      for (var i = 0; i < qcVals.length; i++) {
        var row = qcVals[i];
        var maEM = String(row[0] || "").trim().toUpperCase();  // Cột A (Index 0): Mã đơn EM
        if (!maEM) continue;

        var mspJ = String(row[9] || "").trim();               // Cột J (Index 9): MSP Cũ (Chính xác theo yêu cầu)
        var mspFinal = mspJ;

        var qcV = String(row[21] || "").trim();               // Cột V (Index 21): Qui cách dệt
        var kgAA = parseFloat(row[26]) || 0;                 // Cột AA (Index 26): KG/TẤM BTP

        // Tách khổ mắt / số kèo ở vị trí thứ 3 (sau dấu ' x ') từ chuỗi Cột V (VÙNG ĐỎ KHOANH)
        // Ví dụ: 380/30BX x 30mm x 960 x 505 -> Tách thành phần thứ 3 = 960
        var matMaxVal = 0;
        if (qcV) {
          var partsQC = qcV.split(/\s+[xX]\s+/);
          if (partsQC.length >= 3) {
            var valStr = partsQC[2].replace(/[^0-9\.]/g, "");
            matMaxVal = parseFloat(valStr) || 0;
          } else {
            var nums = qcV.match(/\d+/g);
            if (nums && nums.length >= 2) {
              matMaxVal = parseFloat(nums[nums.length - 2]) || 0;
            }
          }
        }

        if (!mapQC[maEM]) {
          mapQC[maEM] = { maEM: maEM, mspERP: mspFinal, qcV: qcV, kgAA: kgAA, matMax: matMaxVal };
        } else {
          if (matMaxVal > mapQC[maEM].matMax) {
            mapQC[maEM].matMax = matMaxVal;
            if (qcV) mapQC[maEM].qcV = qcV; // Cập nhật chuỗi Qui cách dệt tương ứng với tấm lớn nhất
          }
          if (kgAA > 0) mapQC[maEM].kgAA = kgAA;
          if (mspFinal) mapQC[maEM].mspERP = mspFinal;
        }
      }
    }
  }

  // 2. Đọc dữ liệu quét thô từ Sheet QRBTP (Từ hàng fromRow đến toRow)
  // Quy tắc: 1 QR được tính là 1
  var numRows = toRow - fromRow + 1;
  if (numRows <= 0) return;

  var qrVals = sheetQR.getRange(fromRow, 1, numRows, 5).getValues();
  var gomData = {};

  for (var j = 0; j < qrVals.length; j++) {
    var qrRow = qrVals[j];
    var maEM = String(qrRow[1] || "").trim().toUpperCase();   // Cột B: Mã đơn EM (Mã SX - KEY tra cứu)
    var soKeo = parseFloat(qrRow[2]) || 0;                     // Cột C: Số kèo / Khổ kiện
    if (!maEM || soKeo <= 0) continue;

    if (!gomData[maEM]) {
      gomData[maEM] = {
        maEM: maEM,
        kienMap: {},
        tongKienEyes: 0
      };
    }

    if (!gomData[maEM].kienMap[soKeo]) {
      gomData[maEM].kienMap[soKeo] = 0;
    }
    // 1 QR = 1 kiện
    gomData[maEM].kienMap[soKeo] += 1;
    gomData[maEM].tongKienEyes += soKeo;
  }

  // 3. Tính toán và tạo dòng kết quả xuất sang Sheet 'Tổng Hợp'
  var rowsTongHop = [];
  var emKeys = Object.keys(gomData);

  for (var k = 0; k < emKeys.length; k++) {
    var maEM = emKeys[k];
    var item = gomData[maEM];
    var infoQC = mapQC[maEM] || mapQC[maEM.trim().toUpperCase()] || {};

    var mspERP = infoQC.mspERP || maEM;
    var qcV = infoQC.qcV || "";
    var kgAA = infoQC.kgAA || 0;
    var matTam = infoQC.matMax || 960;

    var soKeoKeys = Object.keys(item.kienMap);
    if (soKeoKeys.length > 0) {
      var tongEyes = item.tongKienEyes;
      var kienNotes = [];
      for (var s = 0; s < soKeoKeys.length; s++) {
        var sk = soKeoKeys[s];
        var countSk = item.kienMap[sk];
        kienNotes.push(countSk + "k " + sk);
      }
      var noteText = kienNotes.join(" + ");

      // CÔNG THỨC CHUẨN XÁC THEO YÊU CẦU:
      // Ratio = tongEyes / matTam (Lúc tính giữ nguyên không làm tròn, VD: 2.6875)
      // Cột C (Số Tấm) = Làm tròn 1 chữ số thập phân khi đưa vào Sheet (VD: 2.6875 -> 2.7)
      // Cột D (Số Kg) = (Ratio chưa làm tròn * kgAA) Làm tròn 1 chữ số thập phân (VD: 2.6875 * 29 = 77.9375 -> 77.9)
      var ratioTam = (matTam > 0) ? (tongEyes / matTam) : 0;
      var soTamLe = Math.round(ratioTam * 10) / 10;
      var soKgLe = Math.round((ratioTam * kgAA) * 10) / 10;

      rowsTongHop.push([
        mspERP,        // Cột A: MSP (Cột J trong QC)
        qcV,           // Cột B: QC (Cột V trong QC)
        soTamLe,       // Cột C: Số Tấm (Số tấm hoàn chỉnh ghép được)
        soKgLe,        // Cột D: Số Kg
        maEM,          // Cột E: Số đơn
        noteText       // Cột F: Số kiện (VD: "5k 166 + 5k 83" hoặc "5k 133")
      ]);
    }
  }

  // 4. Ghi dữ liệu vào Sheet 'Tổng Hợp' (BỎ TIÊU ĐỀ, ĐỔ TỪ HÀNG 6 TRỞ XUỐNG)
  var lastRowTH = sheetTH.getLastRow();
  if (lastRowTH >= 6) {
    sheetTH.getRange(6, 1, lastRowTH - 5, 6).clearContent();
  }

  if (rowsTongHop.length > 0) {
    sheetTH.getRange(6, 1, rowsTongHop.length, 6).setValues(rowsTongHop);
  }
}
