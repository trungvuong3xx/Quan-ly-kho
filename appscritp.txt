// ── Google Apps Script Dành Riêng Cho Quét BTP ────────────────
function doPost(e) {
  var result;
  try {
    var body = JSON.parse(e.postData.contents);
    if (body.action === "luuBTP") {
      result = luuBTP(body.data || body.payload);
    } else {
      result = { success: false, message: "Action không hợp lệ: " + body.action };
    }
  } catch (err) {
    result = { success: false, message: "Lỗi server: " + err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function luuBTP(data) {
  try {
    if (!data || !Array.isArray(data) || data.length === 0) {
      return { success: false, message: "Không có dữ liệu BTP" };
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    // Lấy sheet tên QRBTP hoặc sheet đang chọn hiện tại
    var sheet = ss.getSheetByName("QRBTP") || ss.getActiveSheet();

    // Dữ liệu đổ từ hàng 5 trở đi (không tự thêm dòng tiêu đề)
    var startRow = Math.max(sheet.getLastRow() + 1, 5);

    var rows = data.map(function(item) {
      var ngay = item.ngay || "";
      
      // Định dạng chỉ lấy Giờ:Phút:Giây (HH:mm:ss) chính xác lúc quét
      var gioQuet = "";
      if (item.thoiGian) {
        try {
          var d = new Date(item.thoiGian);
          if (!isNaN(d.getTime())) {
            gioQuet = Utilities.formatDate(d, "GMT+7", "HH:mm:ss");
          } else {
            gioQuet = String(item.thoiGian);
          }
        } catch (e) {
          gioQuet = String(item.thoiGian);
        }
      }

      return [
        item.rawQR || item.id || "",   // Cột A: Mã QR Gốc
        item.msp || "",                // Cột B: Mã BTP
        Number(item.kg || 0),          // Cột C: Số Lượng
        ngay,                          // Cột D: Ngày (YYYY-MM-DD)
        gioQuet                        // Cột E: Giờ quét (HH:mm:ss)
      ];
    });

    // Ghi mảng dữ liệu trực tiếp từ startRow
    sheet.getRange(startRow, 1, rows.length, 5).setValues(rows);

    return { success: true, message: "Đã lưu " + rows.length + " dòng từ hàng " + startRow };
  } catch (err) {
    return { success: false, message: "Lỗi lưu BTP: " + err.message };
  }
}
