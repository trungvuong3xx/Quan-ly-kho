var CONFIG_TONKHO = {
  LIST_SHEET_NAME: 'LIST',
  LIST_COL_TEN: 1,
  LIST_COL_MSP: 2,
  LIST_START_ROW: 2,
  MSP_COL_IN_MONTH_SHEET: 7
};

var VUNG_THUONG_TK_RANGES = [[6, 543], [637, 644]];
var VUNG_FOR_TK_RANGES = [[544, 630]];

// NOTE: doPost được định nghĩa trong file cx5.gs (không được khai báo lại ở đây)

function findRowByMspTonKho_(sheet, msp, ranges) {
  const target = String(msp).trim();
  for (const [start, end] of ranges) {
    const numRows = end - start + 1;
    const values = sheet.getRange(start, CONFIG_TONKHO.MSP_COL_IN_MONTH_SHEET, numRows, 1).getValues();
    for (let i = 0; i < values.length; i++) {
      if (String(values[i][0]).trim() === target) {
        return start + i;
      }
    }
  }
  return -1;
}

function layDanhSachSanPhamTonKhoCX5() {
  try {
    const cacheKey = "tonKhoDanhSach_v1";
    const cache = CacheService.getScriptCache();
    const cached = cache.get(cacheKey);
    if (cached) {
      try { return { list: JSON.parse(cached) }; } catch (e) {}
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const listSheet = ss.getSheetByName(CONFIG_TONKHO.LIST_SHEET_NAME);
    if (!listSheet) return { list: [] };

    const lastRow = listSheet.getLastRow();
    if (lastRow < CONFIG_TONKHO.LIST_START_ROW) return { list: [] };

    const numRows = lastRow - CONFIG_TONKHO.LIST_START_ROW + 1;
    const data = listSheet.getRange(CONFIG_TONKHO.LIST_START_ROW, 1, numRows, 2).getValues();

    const list = data
      .filter(function (r) { return r[CONFIG_TONKHO.LIST_COL_TEN - 1] && r[CONFIG_TONKHO.LIST_COL_MSP - 1]; })
      .map(function (r) {
        return { ten: String(r[CONFIG_TONKHO.LIST_COL_TEN - 1]).trim(), msp: String(r[CONFIG_TONKHO.LIST_COL_MSP - 1]).trim() };
      });

    try { cache.put(cacheKey, JSON.stringify(list), 1800); } catch (e) {}

    return { list: list };
  } catch (err) {
    return { error: err.message };
  }
}

function layTonKhoCX5(payload) {
  try {
    const dateStr = payload && payload.dateStr;
    const msps = (payload && payload.msps) || [];
    if (msps.length === 0) return { list: [] };

    const date = new Date(dateStr);
    if (isNaN(date.getTime())) return { error: "Ngày không hợp lệ" };

    const sheetName = String(date.getMonth() + 1);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { error: 'Chưa có sheet tháng "' + sheetName + '"' };

    const ketQua = [];
    msps.forEach(function (msp) {
      const rowThuong = findRowByMspTonKho_(sheet, msp, VUNG_THUONG_TK_RANGES);
      const rowFor = findRowByMspTonKho_(sheet, msp, VUNG_FOR_TK_RANGES);

      if (rowThuong !== -1) {
        const v = sheet.getRange(rowThuong, 8, 1, 4).getValues()[0];
        ketQua.push({ msp: msp, kho: "Thường", baoDau: v[0] || 0, kgDau: v[1] || 0, baoCuoi: v[2] || 0, kgCuoi: v[3] || 0 });
      }
      if (rowFor !== -1) {
        const v = sheet.getRange(rowFor, 8, 1, 4).getValues()[0];
        ketQua.push({ msp: msp, kho: "FOR", baoDau: v[0] || 0, kgDau: v[1] || 0, baoCuoi: v[2] || 0, kgCuoi: v[3] || 0 });
      }
    });

    return { list: ketQua };
  } catch (err) {
    return { error: err.message };
  }
}
