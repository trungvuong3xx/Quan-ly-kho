function xuLyTaoThangMoi(tenCu, tenMoi) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ui = SpreadsheetApp.getUi();

  if (!tenCu) throw new Error("Chưa nhập tên sheet cũ");
  if (!tenMoi) throw new Error("Chưa nhập tên sheet mới");

  var shCu = ss.getSheetByName(tenCu);
  if (!shCu) throw new Error("Làm gì có cái sheet nào như thế??? " + tenCu);

  var shMoiCheck = ss.getSheetByName(tenMoi);
  if (shMoiCheck) ss.deleteSheet(shMoiCheck);

  var shMoi = shCu.copyTo(ss).setName(tenMoi);
  ss.setActiveSheet(shMoi);

  // Xóa tất cả ghi chú khi tạo tháng mới
  shMoi.getRange(1, 1, shMoi.getMaxRows(), shMoi.getMaxColumns()).clearNote();

  var VUNG_XOA = [
    "AF6:JS299",
    "AF301:JS314",
    "AF319:JS457",
    "AF461:JS536",
    "AF540:JS540",
    "AF544:JS630",
    "AF637:JS664",
    "AF668:JS673",
    "AF677:JS679"
  ];
  shMoi.getRangeList(VUNG_XOA).clearContent();

  var VUNG_HI = [
    [6,   299],
    [301, 314],
    [319, 457],
    [461, 536],
    [540, 540],
    [544, 630],
    [637, 664],
    [668, 673],
    [677, 679]
  ];

  for (var v = 0; v < VUNG_HI.length; v++) {
    var dongDau = VUNG_HI[v][0];
    var dongCuoi = VUNG_HI[v][1];
    var soDong = dongCuoi - dongDau + 1;
    var congThucHI = [];
    for (var i = 0; i < soDong; i++) {
      var dong = dongDau + i;
      congThucHI.push(["='" + tenCu + "'!J" + dong, "='" + tenCu + "'!K" + dong]);
    }
    shMoi.getRange(dongDau, 8, soDong, 2).setFormulas(congThucHI);
  }

  ui.alert("Vạn tuế! Vạn Vạn tuế!", "Khởi bẩm Bệ hạ, Thần đã tạo xong tháng mới 🙇", ui.ButtonSet.OK);
}
