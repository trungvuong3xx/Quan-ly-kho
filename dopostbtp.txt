/**
 * ===================================================================
 * SERVER WEB APP: Nhận dữ liệu BTP, định dạng Giờ chuẩn & Chống trùng
 * Kèm API lấy danh sách QC tính KG tự động cho Bảng Tổng Hợp
 * ===================================================================
 */

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    
    // 1. LƯU DỮ LIỆU QUÉT BTP TỪ THIẾT BỊ
    if (action === "luuBTP") {
      var data = contents.data || [];
      if (!Array.isArray(data) || data.length === 0) {
        return ContentService.createTextOutput(JSON.stringify({ success: true, count: 0 }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName("QRBTP") || ss.getActiveSheet();
      var lastRow = sheet.getLastRow();
      
      // 1. Đọc các dòng gần nhất (tối đa 300 dòng) để kiểm tra chống trùng lặp
      var existingKeys = {};
      if (lastRow >= 5) {
        var startCheckRow = Math.max(5, lastRow - 300);
        var checkCount = lastRow - startCheckRow + 1;
        var existingVals = sheet.getRange(startCheckRow, 1, checkCount, 5).getValues();
        
        for (var k = 0; k < existingVals.length; k++) {
          var exRow = existingVals[k];
          var exID = String(exRow[0] || "").trim();
          var exTime = exRow[4] instanceof Date 
            ? Utilities.formatDate(exRow[4], Session.getScriptTimeZone(), "HH:mm:ss") 
            : String(exRow[4] || "").trim();
          
          if (exID) {
            existingKeys[exID + "|" + exTime] = true;
          }
        }
      }

      var rowsToAppend = [];
      for (var i = 0; i < data.length; i++) {
        var item = data[i];
        var rawID = String(item.rawQR || item.id || "").trim();
        if (!rawID) continue;

        // Cột D (4): Ngày -> Date Object
        var ngayDate;
        if (item.ngay) {
          var p = String(item.ngay).split("-");
          if (p.length === 3) {
            ngayDate = new Date(parseInt(p[0], 10), parseInt(p[1], 10) - 1, parseInt(p[2], 10));
          } else {
            ngayDate = new Date(item.ngay);
          }
        } else {
          ngayDate = new Date();
        }

        // Cột E (5): Thời gian -> Chuỗi giờ chuẩn "HH:mm:ss" (VD: "17:38:51")
        var thoiGianDate = item.thoiGian ? new Date(item.thoiGian) : new Date();
        if (isNaN(thoiGianDate.getTime())) thoiGianDate = new Date();
        
        var thoiGianStr = Utilities.formatDate(thoiGianDate, Session.getScriptTimeZone(), "HH:mm:ss");

        // Key chống trùng lặp
        var key = rawID + "|" + thoiGianStr;

        if (existingKeys[key]) continue; // ĐÃ CÓ TRÊN SHEET -> BỎ QUA BẢN GHI TRÙNG
        existingKeys[key] = true;

        // Cột F (6): Số phiên (1..5)
        var phienVal = item.phien ? parseInt(item.phien, 10) : 1;

        rowsToAppend.push([
          rawID,         // Cột A: ID
          item.msp || "",// Cột B: Đơn
          item.kg || 0,  // Cột C: Loại
          ngayDate,      // Cột D: Ngày
          thoiGianStr,   // Cột E: Thời gian dạng "HH:mm:ss" (VD: 17:38:51)
          phienVal       // Cột F: Phiên
        ]);
      }
      
      if (rowsToAppend.length > 0) {
        var startRow = lastRow < 4 ? 5 : lastRow + 1;
        var numRows = rowsToAppend.length;

        // Ghi dữ liệu
        sheet.getRange(startRow, 1, numRows, 6).setValues(rowsToAppend);

        // Định dạng hiển thị Cột D (Ngày) và Cột E (Giờ hh:mm:ss)
        sheet.getRange(startRow, 4, numRows, 1).setNumberFormat("yyyy-MM-dd");
        sheet.getRange(startRow, 5, numRows, 1).setNumberFormat("hh:mm:ss");
      }
      
      return ContentService.createTextOutput(JSON.stringify({ success: true, count: rowsToAppend.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 2. LẤY DANH MỤC QC ĐỂ TÍNH KG TRÊN BẢNG TỔNG HỢP
    else if (action === "layDanhSachQCBTP" || action === "layDanhSachQC") {
      var ssQC = SpreadsheetApp.getActiveSpreadsheet();
      var sheetQC = ssQC.getSheetByName("QC");
      if (!sheetQC) {
        return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Không tìm thấy Sheet QC" }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var lastRowQC = sheetQC.getLastRow();
      var list = [];
      if (lastRowQC >= 2) {
        var qcVals = sheetQC.getRange(2, 1, lastRowQC - 1, 27).getValues();
        for (var j = 0; j < qcVals.length; j++) {
          var rowQC = qcVals[j];
          var maEM = String(rowQC[0] || "").trim().toUpperCase();
          if (!maEM) continue;
          var qcV = String(rowQC[21] || "").trim();
          var kgAA = parseFloat(rowQC[26]) || 0;
          var tach = tachSoMatVaChieuDaiQC_DoPost(qcV);
          list.push({
            maEM: maEM,
            qcV: qcV,
            mat: tach.mat,
            dai: tach.dai,
            kgAA: kgAA
          });
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, list: list }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    else {
      return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Action không hợp lệ: " + action }))
        .setMimeType(ContentService.MimeType.JSON);
    }

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Hàm phụ trợ tách số mắt và chiều dài từ chuỗi quy cách QC (tên riêng tránh trùng với file tổng hợp)
function tachSoMatVaChieuDaiQC_DoPost(qcV) {
  var matVal = 0;
  var daiVal = 0;
  if (!qcV) return { mat: 0, dai: 0 };
  
  var partsQC = String(qcV).trim().split(/\s+[xX]\s+/);
  if (partsQC.length < 3) {
    partsQC = String(qcV).trim().split(/\s*[xX]\s*/);
  }

  if (partsQC.length >= 3) {
    var valStr = partsQC[2].replace(/[^0-9\.]/g, "");
    matVal = parseFloat(valStr) || 0;
  } else {
    var nums = String(qcV).match(/\d+/g);
    if (nums && nums.length >= 2) {
      matVal = parseFloat(nums[nums.length - 2]) || 0;
    }
  }

  if (partsQC.length >= 4) {
    var daiStr = partsQC[3].replace(/[^0-9\.]/g, "");
    daiVal = parseFloat(daiStr) || 0;
  }

  return { mat: matVal, dai: daiVal };
}
